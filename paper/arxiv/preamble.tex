%
% Requested but deliberately not loaded, with reasons, so the list stays honest:
%
%   shadows          Superseded by the `softshadow' style below, which composes a
%                    soft falloff by hand. TikZ's `drop shadow' is a hard offset
%                    copy and reads as clip-art at figure scale.
%   shadows.blur     Not present in this pgf build (3.1.11a). `softshadow' delivers
%                    the effect without it.
%   shapes.geometric The cylinder was tried for the three datastores in
%                    fig-dataflow. It sizes its body from the node text and then
%                    adds elliptical caps, so a box holding seven lines becomes
%                    enormous and the text escapes it. Right shape for a two-word
%                    label, wrong one for these. Reverted rather than shipped.
%   shapes.misc      Pill shapes for aliases: nothing in the current figures needs a
%                    shape distinct from the card, and adding one would dilute the
%                    role-based vocabulary the key teaches.
%   matrix, chains   The grid and pipeline figures are already exactly aligned by
%                    computed coordinates, and both libraries would replace working
%                    layout with a second mechanism doing the same job. Loading a
%                    library nothing uses costs compile time and misdescribes the
%                    figures, so they are out.
%   graphs, petri    Would suit a state-machine or Petri-net view of queue
%                    arbitration -- a genuinely useful addition, but a new figure
%                    rather than an improvement to an existing one.
%   mindmap          No figure in a systems paper of this kind wants it.
%   graphdrawing     Requires LuaLaTeX. The paper builds under lualatex, so this is
%                    reachable, but automatic layout is worse than computed
%                    placement for block diagrams whose column alignment carries
%                    meaning. Switching the engine would also mean revalidating the
%                    Type 1 font work.

% shadows.blur is not in this pgf build, so the soft shadow is composed by hand:
% three offset copies at low opacity approximate a Gaussian falloff closely enough
% at figure scale, and unlike a hard `drop shadow' it does not read as clip-art.
% Applied through `preaction` so it paints under the node's own fill.
\tikzset{
  softshadow/.style={
    preaction={fill=black, opacity=0.045, transform canvas={xshift=1.7pt, yshift=-1.7pt}},
    preaction={fill=black, opacity=0.045, transform canvas={xshift=1.1pt, yshift=-1.1pt}},
    preaction={fill=black, opacity=0.045, transform canvas={xshift=0.5pt, yshift=-0.5pt}},
  },
  softshadow/.default=,
}

% ------------------------------------------------- the figure type scale ------
% One scale for every figure, so legibility is a single global decision rather
% than a per-diagram accident. The floor is 7pt: below that, figure text stops
% being readable in print and in a two-page-per-sheet review printout, which is
% how most reviewers will see it. Diagram content was trimmed to fit this scale
% rather than the scale shrunk to fit the content.
%
%   \figH  8pt  box headings and band labels
%   \figB  7pt  detail lines inside a box
%   \figE  7pt  edge labels
%
% ABSOLUTE, not relative. This matters and was got wrong once: with \figB defined
% as \scriptsize, the figure type tracked the body size, so moving from the 10pt
% two-column layout to the 11pt single-column one silently enlarged every figure
% label from 7pt to 8pt. The figures' box geometry is fixed in centimetres, so the
% larger text wrapped to more lines than the boxes had room for and overflowed into
% its neighbours -- a failure no overfull-box warning reports, because nothing
% overflows the *page*. Pinning the sizes means a figure renders exactly as it was
% laid out, in either layout. Narrow figures are then scaled up by \cbscaled, which
% is where on-page legibility comes from.
\newcommand{\figH}{\fontsize{8}{9.4}\selectfont}
\newcommand{\figB}{\fontsize{7}{8.3}\selectfont}
\newcommand{\figE}{\fontsize{7}{8.3}\selectfont}

% ------------------------------------------------------------- table system ----
% The tables in this paper are dense and carry the paper's most contested claims,
% so they are built like figures rather than like spreadsheets:
%
%   * A saturated header band with reversed type, which separates the head from
%     the body without a heavy rule.
%   * Zebra banding, so the eye tracks one row across six columns.
%   * Group subheads inside the body, for tables that partition their rows.
%   * Chips instead of dashes. "n/a" set in a bordered chip reads as a deliberate
%     statement; an em dash reads as a gap someone forgot to fill, which is the
%     opposite of what these cells mean.
\newcommand{\thead}[1]{\textcolor{white}{\bfseries #1}}
\newcommand{\theadrow}{\rowcolor{cbInk}}
\newcommand{\zebra}{\rowcolor{cbTableAlt}}

% A group subhead spanning #1 columns.
\newcommand{\tgroup}[2]{%
  \noalign{\vspace{0.35ex}}%
  \rowcolor{cbSurfaceBg}\multicolumn{#1}{@{}>{\bfseries}l@{}}{#2}}

% A chip: a small bordered label for a categorical cell value.
\DeclareRobustCommand{\chip}[2]{%
  \tikz[baseline=(cbchip.base)]{%
    \node[#1, rounded corners=1.8pt, inner xsep=3.2pt, inner ysep=0.9pt,
          line width=0.5pt, font=\footnotesize] (cbchip) {#2};}}
\DeclareRobustCommand{\chipna}{\chip{draw=cbRule, fill=cbMutedBg, text=cbMuted}{n/a}}
\DeclareRobustCommand{\chipnone}{\chip{draw=cbRule, fill=white, text=cbMuted}{none}}
\DeclareRobustCommand{\chipenf}{\chip{draw=cbGovern, fill=cbGovernBg,
  text=cbGovern, line width=0.8pt}{enforced}}
\DeclareRobustCommand{\chipadv}{\chip{draw=cbGovern, fill=white, text=cbGovern,
  densely dashed}{advisory}}
\DeclareRobustCommand{\chipunmeasured}{\chip{draw=cbGovern, fill=cbGovernBg,
  text=cbGovern}{not measured}}
\DeclareRobustCommand{\chipyes}{\chip{draw=cbExtern, fill=cbExternBg, text=cbExtern}{yes}}
\DeclareRobustCommand{\chipstated}{\chip{draw=cbControl, fill=cbControlBg,
  text=cbControl}{stated}}
