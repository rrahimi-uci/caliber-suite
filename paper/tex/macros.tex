%% macros.tex -- controlled vocabulary and quantitative facts.
%%
%% Every recurring term and every number in the paper is defined once here.
%% Two reasons, both of which a reviewer will notice if they are violated:
%%
%%   1. Terminology consistency. The system distinguishes "governed asset" from
%%      "asset family", "candidate-advancement gate" from "release verdict", and
%%      "containment" from "isolation". Those distinctions carry technical
%%      weight, so the paper spells them the same way every time.
%%   2. Numeric single-sourcing. Counts that come from the implementation are
%%      \input from generated/stats.tex, which paper/scripts/gen_stats.py
%%      recomputes from the tree. A number can therefore never drift out of
%%      agreement with the artifact it describes.

% Text-mode separators for figure labels. Using these instead of $\cdot$ and
% $\to$ keeps every glyph in the Times family, so a \tiny figure label does not
% mix a Computer Modern math symbol into a line of Times (see usenix-layout.sty
% on why mathptmx is not loaded).
% \dt is drawn rather than set. A centred dot lives only in TS1, whose coverage
% varies by family, and any miss falls back to a Metafont bitmap that embeds as a
% Type 3 font -- a routine camera-ready rejection. A rule is vector everywhere,
% scales with the surrounding font because it is measured in ex, and is
% indistinguishable from a middot at figure-label sizes.
\newcommand{\dt}{\hspace{0.28em}\raisebox{0.42ex}{\rule{0.30ex}{0.30ex}}\hspace{0.28em}}

% The inline arrow reuses the same Stealth tip as every figure edge, so a stage
% arrow in prose and a stage arrow in a diagram are the same mark.
\newcommand{\arrowto}{\hspace{0.28em}\tikz[baseline=-0.42ex]{%
  \draw[-{Stealth[length=2.7pt,width=2.2pt]}, line width=0.4pt]
    (0,0) -- (0.40,0);}\hspace{0.28em}}

% ------------------------------------------------------------ figure legend ----
% The shared key that lets nine figures be read with one learned vocabulary.
\newcommand{\keyswatch}[2]{%
  \tikz[baseline=-0.55ex]{\node[#1, minimum width=5.5pt, minimum height=5.5pt,
        inner sep=0pt, rounded corners=0.8pt]{};}\,\textnormal{#2}}
% Broken into two explicitly centred rows rather than left to reflow. Relying on
% line breaking makes the key's height depend on the column it lands in, and the
% seven items do not fit one single-column line; fixing the split keeps the
% legend identical under every figure and every column width.
\newcommand{\keyrow}[1]{\mbox{#1}}
\newcommand{\figkey}{%
  \par\vspace{0.5ex}%
  \begingroup\footnotesize\centering
  \keyrow{\keyswatch{surface}{surfaces}}\hspace{0.85em}%
  \keyrow{\keyswatch{control}{control plane}}\hspace{0.85em}%
  \keyrow{\keyswatch{govern}{governance}}\hspace{0.85em}%
  \keyrow{\keyswatch{asset}{asset families}}\par
  \vspace{0.25ex}%
  \keyrow{\keyswatch{extern}{external systems}}\hspace{0.85em}%
  \keyrow{\keyswatch{async}{asynchronous}}\hspace{0.85em}%
  \keyrow{\keyswatch{store}{durable state}}\par
  \endgroup}
